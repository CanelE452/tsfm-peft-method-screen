def _command(args: list[str]) -> dict:
    proc = subprocess.run(args, cwd=ROOT, capture_output=True, text=True, timeout=60)
    return {"args": args, "returncode": proc.returncode, "stdout": proc.stdout.strip(), "stderr": proc.stderr.strip()}


def environment() -> dict:
    import chronos  # noqa: F401
    import peft  # noqa: F401
    import transformers  # noqa: F401

    packages = {}
    for name in ["torch", "chronos-forecasting", "peft", "transformers", "numpy", "pandas", "scipy", "safetensors"]:
        try:
            packages[name] = md.version(name)
        except md.PackageNotFoundError:
            packages[name] = "MISSING"
    return {
        "python": sys.version,
        "executable": sys.executable,
        "platform": platform.platform(),
        "packages": packages,
        "torch_cuda": torch.version.cuda,
        "cuda_available": bool(torch.cuda.is_available()),
        "gpu": torch.cuda.get_device_name(0) if torch.cuda.is_available() else None,
        "vram_bytes": int(torch.cuda.get_device_properties(0).total_memory) if torch.cuda.is_available() else None,
        "nvidia_smi": _command(["nvidia-smi", "--query-gpu=name,memory.total,memory.used,memory.free,driver_version", "--format=csv,noheader"]),
        "source_hashes": source_hashes(),
    }


def _pairs(ids: list[str], origins: np.ndarray) -> list[tuple[str, int]]:
    return [(sid, int(o)) for sid in ids for o in origins]


@torch.no_grad()
def _predict(model: ForecastModel, x: np.ndarray, batch_size: int = 16) -> np.ndarray:
    out = []
    for start in range(0, len(x), batch_size):
        xb = torch.as_tensor(x[start : start + batch_size], device="cuda", dtype=torch.float32)
        q = model(xb)
        if not torch.isfinite(q).all():
            raise RuntimeError("IMPLEMENTATION_FAILURE: nonfinite forecast")
        out.append(q.cpu().numpy())
    return np.concatenate(out, axis=0)


@torch.no_grad()
def parity(model: ForecastModel, d) -> dict:
    pairs = _pairs([DEV_IDS[0]], d.origins("dev", "adapt_a", stride=24)[:2])
    clean_x, _, _, _ = d.batch(pairs, group="dev", role="P0_PARITY", condition="CLEAN")
    miss_x, _, _, _ = d.batch(pairs, group="dev", role="P0_PARITY", condition="IID48", repeat=0)
    records = []
    for label, x in [("clean", clean_x), ("missing_nan_context", miss_x)]:
        direct = _predict(model, x)
        native = model.pipeline.predict(torch.as_tensor(x, dtype=torch.float32), prediction_length=CONFIG["data"]["horizon"]).swapaxes(1, 2).numpy()
        diff = np.abs(direct - native)
        records.append(
            {
                "case": label,
                "shape": list(direct.shape),
                "max_abs": float(diff.max()),
                "allclose": bool(np.allclose(direct, native, rtol=CONFIG["model"]["parity_rtol"], atol=CONFIG["model"]["parity_atol_scaled"])),
            }
        )
    ok = all(r["allclose"] for r in records)
    return {"status": "PASS" if ok else "IMPLEMENTATION_FAILURE", "records": records}


def _score(q: np.ndarray, y: np.ndarray, sigma: np.ndarray) -> float:
    return mean_score(np.sort(q, axis=-1), y, sigma)


def h1_precheck(model: ForecastModel, d) -> dict:
    profiles = {"adapt_a": {}, "adapt_b": {}}
    block_scores = []
    for part in ["adapt_a", "adapt_b"]:
        origins = d.origins("dev", part, stride=24)
        pairs = _pairs(DEV_IDS, origins)
        x, y, sigma, _ = d.batch(pairs, group="dev", role="P0_H1", condition="CLEAN")
        q = _predict(model, x)
        for sid_i, sid in enumerate(DEV_IDS):
            lo, hi = sid_i * len(origins), (sid_i + 1) * len(origins)
            a, b = affine(q[lo:hi], y[lo:hi])
            median = a * q[lo:hi, :, 4] + b
            residual = (y[lo:hi] - median) / sigma[lo:hi, None]
            profiles[part][sid] = residual.mean(axis=0)
            block_scores.append(
                {
                    "block": part,
                    "series": sid,
                    "origins": int(len(origins)),
                    "raw_score": _score(q[lo:hi], y[lo:hi], sigma[lo:hi]),
                    "affine_a": float(a),
                    "affine_b": float(b),
                }
            )
    centered = {}
    for part in ["adapt_a", "adapt_b"]:
        mat = np.stack([profiles[part][sid] for sid in DEV_IDS])
        centered[part] = mat - mat.mean(axis=0, keepdims=True)
    rows = []
    passing = 0
    for idx, sid in enumerate(DEV_IDS):
        a = centered["adapt_a"][idx]
        b = centered["adapt_b"][idx]
        denom = float(np.linalg.norm(a) * np.linalg.norm(b))
        cosine = float(np.dot(a, b) / denom) if denom else 0.0
        rms_a = float(np.sqrt(np.mean(a * a)))
        rms_b = float(np.sqrt(np.mean(b * b)))
        ok = cosine > CONFIG["h1"]["precheck_cosine_strict_min"] and rms_a >= CONFIG["h1"]["precheck_min_profile_rms"] and rms_b >= CONFIG["h1"]["precheck_min_profile_rms"]
        passing += int(ok)
        rows.append({"series": sid, "cosine": cosine, "rms_a": rms_a, "rms_b": rms_b, "passes": bool(ok)})
    status = "PASS" if passing >= CONFIG["h1"]["precheck_min_clients"] else "STOP_SETUP_H1_NO_STABLE_HETEROGENEITY"
    return {"status": status, "passing_clients": passing, "client_checks": rows, "block_scores": block_scores}


def h2_precheck(model: ForecastModel, d) -> dict:
    origins = d.origins("dev", "adapt", stride=24)
    pairs = _pairs(DEV_IDS, origins)
    clean_x, y, sigma, _ = d.batch(pairs, group="dev", role="P0_H2", condition="CLEAN")
    block_x, _, _, block_masks = d.batch(pairs, group="dev", role="P0_H2", condition="BLOCK48", repeat=0)
    interp_x, _, _, _ = d.batch(pairs, group="dev", role="P0_H2", condition="BLOCK48", repeat=0, interpolate=True)
    clean_q = _predict(model, clean_x)
    block_q = _predict(model, block_x)
    interp_q = _predict(model, interp_x)
    clean_score = _score(clean_q, y, sigma)
    block_score = _score(block_q, y, sigma)
    interp_score = _score(interp_q, y, sigma)
    block_degradation = 100.0 * (block_score - clean_score) / clean_score
    interp_gap = 100.0 * (interp_score - clean_score) / clean_score
    patch_counts = block_masks.reshape(len(block_masks), -1, CONFIG["data"]["native_patch"]).sum(axis=2)
    partial = ((patch_counts > 0) & (patch_counts < CONFIG["data"]["native_patch"])).sum(axis=1)
    status = "PASS"
    reasons = []
    if block_degradation < CONFIG["h2"]["precheck_block_degradation_pct"]:
        status = "STOP_SETUP_H2_NO_UNRESOLVED_MASK_GAP"
        reasons.append("BLOCK48 degradation below threshold")
    if interp_gap < CONFIG["h2"]["precheck_interpolated_gap_pct"]:
        status = "STOP_SETUP_H2_NO_UNRESOLVED_MASK_GAP"
        reasons.append("interpolation leaves too little unresolved gap")
    if float(partial.mean()) < CONFIG["h2"]["precheck_mean_partial_patch_min"]:
        status = "STOP_SETUP_H2_NO_UNRESOLVED_MASK_GAP"
        reasons.append("too few partial patches for set-position correction")
    return {
        "status": status,
        "reasons": reasons,
        "origins_per_series": int(len(origins)),
        "scores": {
            "clean": clean_score,
            "block48_native": block_score,
            "block48_interp": interp_score,
            "block48_degradation_pct_vs_clean": float(block_degradation),
            "interp_gap_pct_vs_clean": float(interp_gap),
        },
        "mask": {
            "mean_partial_patches": float(partial.mean()),
            "min_partial_patches": int(partial.min()),
            "max_partial_patches": int(partial.max()),
        },
    }


def trainable_inventory() -> dict:
    from peft import LoraConfig, get_peft_model
    from types import MethodType

    def embeddings(self):
        return self.shared

    probe = ForecastModel()
    base = probe.network
    base.get_input_embeddings = MethodType(embeddings, base)
    wrapped = get_peft_model(base, LoraConfig(r=8, lora_alpha=16, lora_dropout=0.0, target_modules=["q", "v"], bias="none"))
    rows = [{"name": n, "shape": list(p.shape), "numel": int(p.numel())} for n, p in wrapped.named_parameters() if p.requires_grad]
    total = int(sum(r["numel"] for r in rows))
    del wrapped, probe
    torch.cuda.empty_cache()
    return {"target_modules": ["q", "v"], "trainable_parameter_count": total, "parameters": rows[:10], "parameter_rows_total": len(rows)}


def run() -> dict:
    setup()
    RESULTS.mkdir(parents=True, exist_ok=True)
    start = time.perf_counter()
    d_audit = data_audit()
    env = environment()
    if not env["cuda_available"]:
        payload = {"status": "RESOURCE_BLOCK", "reason": "CUDA unavailable", "environment": env, "data": d_audit}
        save(RESULTS / "PREFLIGHT.json", payload)
        return payload
    d = load_data()
    model = ForecastModel()
    checks = {
        "environment": env,
        "data": d_audit,
        "parity": parity(model, d),
        "lora_inventory": trainable_inventory(),
    }
    if checks["parity"]["status"] != "PASS":
        payload = {"status": "IMPLEMENTATION_FAILURE", "checks": checks, "elapsed_seconds": time.perf_counter() - start}
        save(RESULTS / "PREFLIGHT.json", payload)
        return payload
    checks["h1"] = h1_precheck(model, d)
    checks["h2"] = h2_precheck(model, d)
    h1_open = checks["h1"]["status"] == "PASS"
    h2_open = checks["h2"]["status"] == "PASS"
    status = "PASS" if (h1_open or h2_open) else "STOP_SETUP_NO_CANDIDATE_OPEN"
    payload = {
        "status": status,
        "candidate_open": {"h1": h1_open, "h2": h2_open},
        "checks": checks,
        "elapsed_seconds": time.perf_counter() - start,
        "test_scores_accessed": False,
    }
    save(RESULTS / "PREFLIGHT.json", payload)
    return payload


if __name__ == "__main__":
    try:
        out = run()
        print(json.dumps(out, indent=2, ensure_ascii=False)[:6000])
    except Exception as exc:
        payload = {"status": "IMPLEMENTATION_FAILURE", "error": repr(exc), "traceback_file": str((RESULTS / "PREFLIGHT_TRACEBACK.txt").relative_to(ROOT))}
        RESULTS.mkdir(parents=True, exist_ok=True)
        import traceback

        (RESULTS / "PREFLIGHT_TRACEBACK.txt").write_text(traceback.format_exc(), encoding="utf-8")
        save(RESULTS / "PREFLIGHT.json", payload)
        raise
